cmake_minimum_required(VERSION 3.16)
project(agv_map_3d LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(cartographer REQUIRED)
find_package(PCL REQUIRED COMPONENTS common io filters)
find_package(Boost 1.71 REQUIRED COMPONENTS system filesystem)
find_package(Threads REQUIRED)
find_package(nlohmann_json REQUIRED)
add_executable(pbstream_to_ply src/pbstream_to_ply.cpp)
target_link_libraries(pbstream_to_ply PRIVATE cartographer)
add_executable(pcd_to_ply src/pcd_to_ply.cpp)
target_include_directories(pcd_to_ply PRIVATE ${PCL_INCLUDE_DIRS})
target_link_directories(pcd_to_ply PRIVATE ${PCL_LIBRARY_DIRS})
target_link_libraries(pcd_to_ply PRIVATE ${PCL_LIBRARIES})
target_compile_options(pcd_to_ply PRIVATE ${PCL_DEFINITIONS})
add_executable(map_3d_server src/map_3d_server.cpp src/assets.cpp)
target_include_directories(map_3d_server PRIVATE include)
target_link_libraries(map_3d_server PRIVATE Boost::system Boost::filesystem Threads::Threads nlohmann_json::nlohmann_json)
foreach(target pbstream_to_ply pcd_to_ply map_3d_server)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
endforeach()
install(TARGETS pbstream_to_ply pcd_to_ply map_3d_server DESTINATION lib/${PROJECT_NAME})
install(FILES README.md DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  enable_testing()
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_executable(map3d_test_fixture tests/make_fixture.cpp)
  target_link_libraries(map3d_test_fixture PRIVATE cartographer)
  add_test(NAME map3d_export_coordinates
    COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_export.py
      $<TARGET_FILE:map3d_test_fixture> $<TARGET_FILE:pbstream_to_ply>)
  add_test(NAME map3d_pcd_export
    COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_pcd_export.py $<TARGET_FILE:pcd_to_ply>)
  add_test(NAME map3d_asset_api
    COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_asset_server.py $<TARGET_FILE:map_3d_server>)
  set_tests_properties(map3d_asset_api PROPERTIES TIMEOUT 120)
endif()
ament_package()
