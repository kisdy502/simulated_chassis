cmake_minimum_required(VERSION 3.8)
project(simulated_chassis_ur5e)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_action REQUIRED)
find_package(control_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(trajectory_msgs REQUIRED)
find_package(controller_manager_msgs REQUIRED)
find_package(agv_bridge_v2_interfaces REQUIRED)
find_package(urdf REQUIRED)
set(ARM_ADAPTER_DEPS rclcpp sensor_msgs trajectory_msgs controller_manager_msgs agv_bridge_v2_interfaces urdf)
add_library(arm_move_adapter_core STATIC src/arm_move_adapter.cpp)
target_include_directories(arm_move_adapter_core PUBLIC include)
ament_target_dependencies(arm_move_adapter_core ${ARM_ADAPTER_DEPS})
add_executable(arm_move_adapter src/arm_move_adapter_main.cpp)
target_link_libraries(arm_move_adapter arm_move_adapter_core)
ament_target_dependencies(arm_move_adapter ${ARM_ADAPTER_DEPS})
install(TARGETS arm_move_adapter DESTINATION lib/${PROJECT_NAME})
add_executable(gripper_control src/gripper_control.cpp)
ament_target_dependencies(gripper_control rclcpp rclcpp_action control_msgs std_srvs std_msgs sensor_msgs trajectory_msgs)
install(TARGETS gripper_control DESTINATION lib/${PROJECT_NAME})
ament_python_install_package(${PROJECT_NAME})
install(DIRECTORY launch urdf config DESTINATION share/${PROJECT_NAME})
install(PROGRAMS scripts/wait_for_robot.py DESTINATION lib/${PROJECT_NAME} RENAME wait_for_robot)
install(PROGRAMS scripts/wrist_camera_node.py DESTINATION lib/${PROJECT_NAME} RENAME wrist_camera_node)
install(FILES README.md DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  find_package(moveit_core REQUIRED)
  add_executable(check_gripper_planning test/check_gripper_planning.cpp)
  ament_target_dependencies(check_gripper_planning moveit_core urdf)
  add_test(NAME test_tool_models COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/test/test_tool_models.py)
  add_test(NAME test_gripper_control COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/test/test_gripper_control.py)
  set_tests_properties(test_tool_models PROPERTIES TIMEOUT 90
    ENVIRONMENT "GRIPPER_PLANNING_CHECK=$<TARGET_FILE:check_gripper_planning>")
  set_tests_properties(test_gripper_control PROPERTIES TIMEOUT 45
    ENVIRONMENT "ROS_DOMAIN_ID=187;RMW_IMPLEMENTATION=rmw_cyclonedds_cpp")
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_arm_move_adapter test/test_arm_move_adapter.cpp
    ENV ROS_DOMAIN_ID=88 RMW_IMPLEMENTATION=rmw_cyclonedds_cpp TIMEOUT 60)
  target_link_libraries(test_arm_move_adapter arm_move_adapter_core)
  ament_target_dependencies(test_arm_move_adapter ${ARM_ADAPTER_DEPS})
endif()
ament_package()
