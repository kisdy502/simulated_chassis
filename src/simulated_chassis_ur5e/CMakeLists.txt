cmake_minimum_required(VERSION 3.8)
project(simulated_chassis_ur5e)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(trajectory_msgs REQUIRED)
find_package(controller_manager_msgs REQUIRED)
find_package(agv_bridge_v2_interfaces REQUIRED)
find_package(urdf REQUIRED)
set(ARM_ADAPTER_DEPS rclcpp sensor_msgs trajectory_msgs controller_manager_msgs agv_bridge_v2_interfaces urdf)
add_library(arm_move_adapter_core STATIC src/arm_move_adapter.cpp)
target_include_directories(arm_move_adapter_core PUBLIC include)
ament_target_dependencies(arm_move_adapter_core ${ARM_ADAPTER_DEPS})
add_executable(arm_move_adapter src/arm_move_adapter_main.cpp)
target_link_libraries(arm_move_adapter arm_move_adapter_core)
ament_target_dependencies(arm_move_adapter ${ARM_ADAPTER_DEPS})
install(TARGETS arm_move_adapter DESTINATION lib/${PROJECT_NAME})
ament_python_install_package(${PROJECT_NAME})
install(DIRECTORY launch urdf config DESTINATION share/${PROJECT_NAME})
install(PROGRAMS scripts/wait_for_robot.py DESTINATION lib/${PROJECT_NAME} RENAME wait_for_robot)
install(PROGRAMS scripts/wrist_camera_node.py DESTINATION lib/${PROJECT_NAME} RENAME wrist_camera_node)
install(FILES README.md DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_arm_move_adapter test/test_arm_move_adapter.cpp
    ENV ROS_DOMAIN_ID=88 RMW_IMPLEMENTATION=rmw_cyclonedds_cpp TIMEOUT 60)
  target_link_libraries(test_arm_move_adapter arm_move_adapter_core)
  ament_target_dependencies(test_arm_move_adapter ${ARM_ADAPTER_DEPS})
endif()
ament_package()
