cmake_minimum_required(VERSION 3.8)
project(battery_simulator)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(std_srvs REQUIRED)
add_executable(battery_simulator_node src/battery_simulator_node.cpp)
target_include_directories(battery_simulator_node PRIVATE include)
ament_target_dependencies(battery_simulator_node rclcpp sensor_msgs nav_msgs geometry_msgs std_msgs std_srvs)
install(TARGETS battery_simulator_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_battery_model test/test_battery_model.cpp)
  target_include_directories(test_battery_model PRIVATE include)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME test_battery_node COMMAND ${Python3_EXECUTABLE}
    ${CMAKE_CURRENT_SOURCE_DIR}/test/test_battery_node.py $<TARGET_FILE:battery_simulator_node>)
  set_tests_properties(test_battery_node PROPERTIES TIMEOUT 40
    ENVIRONMENT "ROS_DOMAIN_ID=90;RMW_IMPLEMENTATION=rmw_cyclonedds_cpp")
endif()
ament_package()
